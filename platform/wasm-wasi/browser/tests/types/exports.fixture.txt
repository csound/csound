const messageListener = (message: string) => void message;
const messageContext = {};
const messageCount: number = browserCsound.listenerCount("message");
browserCsound.on("message", messageListener, messageContext).once("play", () => {});
browserCsound.addListener("message", messageListener, messageContext);
browserCsound.once("message", messageListener, messageContext);
browserCsound.off("message", messageListener, messageContext, true);
browserCsound.removeListener("message");
browserCsound.removeAllListeners("message");
browserCsound.removeAllListeners();
// @ts-expect-error listenerCount requires an event name.
browserCsound.listenerCount();
// @ts-expect-error removeAllListeners accepts one event, not an array.
browserCsound.removeAllListeners(["play", "pause"]);
void messageCount;

const promptEvent: ReadlineEvent = { requestId: 1, prompt: "" };
const closedEvent: ReadlineEvent = { requestId: 1, prompt: null };
browserCsound.on("readline", (event: ReadlineEvent) => { void event; });
void browserCsound.readlineSubmit(promptEvent.requestId, "answer");
void localLibCsound.csoundReadlineSubmit(1, closedEvent.requestId, "answer");
void libcsound({ onReadline(event) { const id: number = event.csound; void id; } });
