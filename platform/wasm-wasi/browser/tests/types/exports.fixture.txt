async function numericChannelValue() {
  const csound = await createCsound();
  const value: number = await csound.getControlChannel("gain");
  await csound.setControlChannel("gain", value);
}

async function liveCodeValue() {
  const csound = await createCsound();
  await csound.setStringChannel("source", "return 42");
  const source: string = await csound.getStringChannel("source");
  const result: number = await csound.evalCode(source);
  const compiled: number = await csound.compileOrc("instr 1\nendin");
  return { result, compiled };
}

import { Csound as createCsound64, libcsound as libcsound64 } from "@csound/browser/wasm64.js";
async function sampleWidths() {
  const normal = await libcsound();
  const wide = await libcsound64();
  const floatTable: Float32Array | undefined = normal.csoundGetTable(0, 1);
  const doubleTable: Float64Array | undefined = wide.csoundGetTable(0, 1);
  const normalBrowser = await createCsound();
  const wideBrowser = await createCsound64();
  const browserFloatTable: Float32Array | undefined = await normalBrowser.getTable("1");
  const browserDoubleTable: Float64Array | undefined = await wideBrowser.getTable("1");
  return [floatTable, doubleTable, browserFloatTable, browserDoubleTable];
}
